cmake_minimum_required(VERSION 3.20)
project(x20ctl_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(x20ctl_core STATIC src/x20_codec.cpp)
target_include_directories(x20ctl_core PUBLIC include)
if(MSVC)
  target_compile_options(x20ctl_core PRIVATE /W4 /permissive- /utf-8)
endif()
add_executable(x20ctl_core_tests tests/core_tests.cpp)
target_link_libraries(x20ctl_core_tests PRIVATE x20ctl_core)
enable_testing()
add_test(NAME native_parity COMMAND x20ctl_core_tests "${CMAKE_SOURCE_DIR}/../artifacts/product-engine-phase/reference-vectors.tsv")
if(WIN32)
  add_executable(x20ctl-engine src/engine_host.cpp)
  target_link_libraries(x20ctl-engine PRIVATE x20ctl_core windowsapp xinput)
  target_compile_definitions(x20ctl-engine PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(x20ctl-engine PRIVATE /W4 /permissive- /utf-8)
endif()
